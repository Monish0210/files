
1.  Program       → Function

2.  Function      → Type main ( ) Block

3.  Type          → int | void

4.  Block         → { Statements }

5.  Statements    → Statement Statements | ε

6.  Statement     → Declaration ;
                  | Assignment ;
                  | IfElse
                  | DoWhile
                  | return Expression ;

7.  Declaration   → int id
                  | int id = Expression

8.  Assignment    → id = Expression

9.  IfElse        → if ( Condition ) Block else Block

10. DoWhile       → do Block while ( Condition ) ;

11. Condition     → Expression > Expression
                  | Expression < Expression

12. Expression    → Expression + Term
                  | Expression - Term
                  | Term

13. Term          → id
                  | number
                  | ( Expression )
