int main() {
    int value = 12;

    if (value > 10) {
        value = value - 2;
    } else {
        value = value + 2;
    }

    do {
        value = value - 1;
    } while (value > 0);

    return 0;
}





1.  Program       → Function

2.  Function      → int main ( ) Block

3.  Block         → { Statements }

4.  Statements    → Statement Statements | ε

5.  Statement     → Declaration ;
                  | Assignment ;
                  | IfElse
                  | DoWhile
                  | return number ;

6.  Declaration   → int id
                  | int id = Expression

7.  Assignment    → id = Expression

8.  IfElse        → if ( Condition ) Block else Block

9.  DoWhile       → do Block while ( Condition ) ;

10. Condition     → Expression > Expression
                  | Expression < Expression

11. Expression    → Expression + Term
                  | Expression - Term
                  | Term

12. Term          → id
                  | number
                  | ( Expression )
