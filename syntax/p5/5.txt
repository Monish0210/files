%{
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

int yylex();
void yyerror(char *s);
%}

%union
{
    double dval;
}

%token <dval> NUMBER

%token SQRT SIN COS TAN LOG PI

%type <dval> expr
%type <dval> term
%type <dval> power
%type <dval> unary
%type <dval> factor

%left '+' '-'
%left '*' '/' '%'
%right '^'
%right UMINUS UPLUS

%%

input:
        /* empty */
        | input line
        ;

line:
        '
'
        |
        expr '
'
        {
            printf("Result = %g

", $1);
            printf("Enter expression (Ctrl+D to quit): ");
        }
        |
        error '
'
        {
            printf("Syntax Error: Invalid expression. Skipping line.

");
            yyerrok;
            printf("Enter expression (Ctrl+D to quit): ");
        }
        ;

expr:
        expr '+' term
        {
            $$ = $1 + $3;
        }
        |
        expr '-' term
        {
            $$ = $1 - $3;
        }
        |
        term
        {
            $$ = $1;
        }
        ;

term:
        term '*' power
        {
            $$ = $1 * $3;
        }
        |
        term '/' power
        {
            if($3 == 0)
            {
                printf("Error: Division by zero!
");
                $$ = 0;
            }
            else
            {
                $$ = $1 / $3;
            }
        }
        |
        term '%' power
        {
            if($3 == 0)
            {
                printf("Error: Modulus by zero!
");
                $$ = 0;
            }
            else
            {
                $$ = fmod($1, $3);
            }
        }
        |
        power
        {
            $$ = $1;
        }
        ;

power:
        unary '^' power
        {
            $$ = pow($1, $3);
        }
        |
        unary
        {
            $$ = $1;
        }
        ;

unary:
        '-' unary %prec UMINUS
        {
            $$ = -$2;
        }
        |
        '+' unary %prec UPLUS
        {
            $$ = $2;
        }
        |
        factor
        {
            $$ = $1;
        }
        ;

factor:
        '(' expr ')'
        {
            $$ = $2;
        }
        |
        NUMBER
        {
            $$ = $1;
        }
        |
        PI
        {
            $$ = M_PI;
        }
        |
        SQRT '(' expr ')'
        {
            if($3 < 0)
            {
                printf("Error: Square root of negative number!
");
                $$ = 0;
            }
            else
            {
                $$ = sqrt($3);
            }
        }
        |
        SIN '(' expr ')'
        {
            $$ = sin($3);
        }
        |
        COS '(' expr ')'
        {
            $$ = cos($3);
        }
        |
        TAN '(' expr ')'
        {
            $$ = tan($3);
        }
        |
        LOG '(' expr ')'
        {
            if($3 <= 0)
            {
                printf("Error: Logarithm requires a positive value!
");
                $$ = 0;
            }
            else
            {
                $$ = log($3);
            }
        }
        ;

%%

void yyerror(char *s)
{
    printf("Syntax Error: Invalid expression
");
}

int main()
{
    printf("Enter expression (Ctrl+D to quit): ");

    yyparse();

    printf("
Goodbye!
");

    return 0;
}
